    <script>
        document.addEventListener('DOMContentLoaded', () => {
            
            /* --- 1. Interacción del Menú Móvil --- */
            const btnMenu = document.getElementById('mobile-menu-btn');
            const mobileMenu = document.getElementById('mobile-menu');
            const menuLinks = mobileMenu.querySelectorAll('a');

            if(btnMenu && mobileMenu) {
                btnMenu.addEventListener('click', () => {
                    mobileMenu.classList.toggle('hidden');
                    const icon = btnMenu.querySelector('i');
                    if(mobileMenu.classList.contains('hidden')) {
                        icon.classList.remove('fa-times');
                        icon.classList.add('fa-bars');
                    } else {
                        icon.classList.remove('fa-bars');
                        icon.classList.add('fa-times');
                    }
                });

                menuLinks.forEach(link => {
                    link.addEventListener('click', () => {
                        mobileMenu.classList.add('hidden');
                        const icon = btnMenu.querySelector('i');
                        icon.classList.remove('fa-times');
                        icon.classList.add('fa-bars');
                    });
                });
            }

            /* --- 2. Smooth Scrolling con ajuste para Navbar Sticky --- */
            document.querySelectorAll('a[href^="#"]').forEach(anchor => {
                anchor.addEventListener('click', function (e) {
                    const targetId = this.getAttribute('href');
                    if (targetId === '#') return;
                    
                    const targetElement = document.querySelector(targetId);
                    if (targetElement) {
                        e.preventDefault();
                        const headerOffset = 96; 
                        const elementPosition = targetElement.getBoundingClientRect().top;
                        const offsetPosition = elementPosition + window.pageYOffset - headerOffset;
      
                        window.scrollTo({
                            top: offsetPosition,
                            behavior: "smooth"
                        });
                    }
                });
            });

            /* --- 3. Validación de Tamaño Máximo de Archivos Adjuntos (10MB) --- */
            const fileInput = document.getElementById('adjuntos');
            const fileErrorMsg = document.getElementById('error-archivo');
            const submitBtn = document.getElementById('btn-enviar');
            const MAX_BYTES = 10 * 1024 * 1024; 

            if(fileInput && fileErrorMsg && submitBtn) {
                fileInput.addEventListener('change', function() {
                    let totalSize = 0;
                    for (let i = 0; i < this.files.length; i++) {
                        totalSize += this.files[i].size;
                    }
                    if (totalSize > MAX_BYTES) {
                        fileErrorMsg.classList.remove('hidden');
                        submitBtn.disabled = true;
                        submitBtn.classList.add('opacity-50', 'cursor-not-allowed');
                        this.value = ''; 
                    } else {
                        fileErrorMsg.classList.add('hidden');
                        submitBtn.disabled = false;
                        submitBtn.classList.remove('opacity-50', 'cursor-not-allowed');
                    }
                });
            }
        });

        /* --- 4. Lógica de Pestañas para Pasarela QR --- */
        function openQR(bankId, element) {
            const contents = document.querySelectorAll('.qr-tab-content');
            contents.forEach(content => {
                content.classList.remove('active');
            });

            const buttons = document.querySelectorAll('.qr-tab-btn');
            buttons.forEach(btn => {
                btn.classList.remove('bg-axon-navy', 'text-white', 'shadow-md');
                btn.classList.add('bg-white', 'text-axon-navy', 'shadow-sm');
            });

            document.getElementById(bankId).classList.add('active');
            element.classList.remove('bg-white', 'text-axon-navy', 'shadow-sm');
            element.classList.add('bg-axon-navy', 'text-white', 'shadow-md');
        }
    </script>
